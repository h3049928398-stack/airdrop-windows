cmake_minimum_required(VERSION 3.24)
project(AirDropUserStack C CXX)
set(CMAKE_CXX_STANDARD 20)
set(LWIP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../upstream/lwip")
include("${LWIP_DIR}/src/Filelists.cmake")
add_library(air_lwip STATIC ${lwipcore_SRCS} ${lwipcore6_SRCS})
target_include_directories(air_lwip PUBLIC include "${LWIP_DIR}/src/include")
target_compile_definitions(air_lwip PRIVATE _CRT_SECURE_NO_WARNINGS)
add_executable(AirDrop.Network main.cpp)
target_link_libraries(AirDrop.Network PRIVATE air_lwip ws2_32)
target_compile_definitions(AirDrop.Network PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
